Crear un Dockerfile con lo siguiente: (en cualquier carpeta de Linux)


----- (guardar en archivo llamado "Dockerfile" -sin extensión-)

FROM alpine:latest

RUN apk update && apk add --no-cache \
    openssh-client \
    bash \
    iproute2 \
    iputils

# Contraseña de root
RUN echo "root:gns3" | chpasswd

VOLUME ["/etc/network", "/root"]
CMD ["/bin/bash"]

------

# Desde la consola 
# (dentro de la carpeta donde se encuentra el Dockerfile)
docker build -t alpine-sshclient .

# EN ALPINE-SSHCLIENT:
# Obliga a Alpine a usar algoritmos de intercambio de contraseña más viejos (son los que usa la imagen vIOS de Cisco)

mkdir -p ~/.ssh

# 192.168.90.1 es el default gateway de la VLAN MGMT
# 192.168.90.28 fue la IP histórica de PVE-QDEVICE; el nodo está detenido
# desde S2.2 y no debe usarse como destino operativo.
# 192.168.90.29 queda sin uso operativo documentado en esta replanificación.

cat << EOF > ~/.ssh/config
Host 192.168.90.1
    KexAlgorithms +diffie-hellman-group14-sha1
    HostKeyAlgorithms +ssh-rsa
Host 192.168.90.28
    # Host histórico de QDevice; conservar sólo para antecedentes/rollback.
    KexAlgorithms +diffie-hellman-group14-sha1
    HostKeyAlgorithms +ssh-rsa
Host 192.168.90.29
    KexAlgorithms +diffie-hellman-group14-sha1
    HostKeyAlgorithms +ssh-rsa
EOF

chmod 600 ~/.ssh/config
