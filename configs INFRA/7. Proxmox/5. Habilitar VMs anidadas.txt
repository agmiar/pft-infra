# Recargar modulo de KVM de virtualizacion anidada
# (en el host Ubuntu de mi PC fisico)
# reemplazar intel por amd, si aplica
# CORRERLO CON GNS3 CERRADO
sudo modprobe -r kvm_intel
sudo modprobe kvm_intel nested=1

# (Debe retornar Y o 1) --> Nested virtualization
cat /sys/module/kvm_intel/parameters/nested

# SI devuelve Y o 1:
echo "options kvm_intel nested=1" | sudo tee /etc/modprobe.d/kvm-intel.conf

Dentro del appliance de Proxmox (click derecho --> Configure)
Ir a la pestaña "Advanced" y agregar esto en "Options"
	"-cpu host,kvm=on,+vmx" (sin las comillas)

(en Proxmox)
modprobe kvm_intel
ls -l /dev/kvm
