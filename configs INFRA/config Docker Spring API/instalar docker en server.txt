encender la imagen Ubuntu Server QEMU
conectar la nube NAT al server

sudo nano /etc/netplan/00-installer-config.yaml

network:
  version: 2
  bonds:
    bond0: # esto es dentro de la red LAN de GNS3
      interfaces:
        - ens3
        - ens4
        - ens5
      addresses:
        - 192.168.50.11/29
      routes:
        - to: default
          via: 192.168.50.1
      parameters:
        mode: active-backup
        primary: ens3
# para detectar caida rápida del enlace
        mii-monitor-interval: 100 
# para que los switches Core limpien la MAC del switch anterior al que estaba conectado el server
        gratuitous-arp: 5
        arp-all-targets: any 

  # para especificar las direcciones MAC de cada interfaz
  ethernets:
    ens3:
      match:
        macaddress: 02:11:22:33:44:11 
    ens4:
      match:
        macaddress: 02:11:22:33:44:12
      set-name: ens4
      accept-ra: true
    ens5:
      match:
        macaddress: 02:11:22:33:44:13
      dhcp4: true # le permite agarrar IP desde la nube NAT
      set-name: ens5
      accept-ra: true
    ens6:
      accept-ra: true
  version: 2

Ctrl + X --> Y --> Enter

sudo netplan apply
